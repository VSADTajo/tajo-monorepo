# Implementation Plan: SFTP Auth Integration

**Branch**: `005-sftp-auth-integration` | **Date**: 2026-03-10 | **Spec**: [spec.md](./spec.md)
**Input**: Feature specification from `/specs/005-sftp-auth-integration/spec.md`

## Summary

Add login and full authentication to the SFTP Mapping Engine. The React SPA gains a login screen that calls `ms-authentication-api` directly for JWT issuance and token refresh. The NestJS backend gains a global `AuthMiddleware` that validates every request's JWT by calling `ms-authentication-api/auth/validate` — the same remote-validation pattern already used by `ms-gateway-api`. Tokens are stored in `localStorage`; a `fetchWithAuth` utility handles automatic token injection and silent refresh on 401.

**Primary Dependencies**: NestJS 11 + `@nestjs/cqrs` 11 + TypeORM 0.3 (backend) / React 19 + React Router 7 + TanStack Query 5 + React Hook Form 7 + Tailwind 4 (frontend)
**Storage**: PostgreSQL 16 (existing — no new tables); `localStorage` (browser tokens)
**Testing**: Jest + Supertest + Testcontainers (backend e2e); no frontend tests in scope
**Target Platform**: Linux server (Hetzner, Ubuntu 24.04) / Browser (Chrome/Firefox)
**Project Type**: Web application (NestJS backend + React SPA frontend)
**Performance Goals**: Auth middleware adds ~1 HTTP call per request to `ms-authentication-api` (same server, LAN latency < 5ms)
**Constraints**: No new DB migrations; no new npm packages on backend; `VITE_AUTH_API_URL` must be set at Docker build time
**Scale/Scope**: Single server deployment; ~10 concurrent users (internal tool)

### Backend: Infrastructure Service

```typescript
// infrastructure/services/auth-validation.service.ts
@Injectable()
export class AuthValidationService implements IAuthValidationService {
    constructor(
        @Inject(Logger) private readonly logger: Logger,
        private readonly authApiConfig: AuthApiConfigService,
    ) {}

    async validate(token: string): Promise<AuthValidationResult> {
        try {
            const response = await axios.post(
                `${this.authApiConfig.authApiUrl}/auth/validate`,
                { token },
            );
            return { valid: response.data?.valid === true, userId: response.data?.userId, roles: response.data?.roles };
        } catch (error) {
            this.logger.error(`Auth validation request failed: ${error}`);
            throw new ProviderError('Authentication service unavailable');
        }
    }
}
```

### Backend: Auth Middleware

```typescript
// infrastructure/middleware/auth.middleware.ts
@Injectable()
export class AuthMiddleware implements NestMiddleware {
    constructor(
        @Inject(Logger) private readonly logger: Logger,
        private readonly authValidationService: IAuthValidationService,
    ) {}

    async use(req: Request, _res: Response, next: NextFunction): Promise<void> {
        const authHeader = req.headers.authorization;
        if (!authHeader?.startsWith('Bearer ')) {
            throw new HttpException('Missing or invalid Authorization header', HttpStatus.UNAUTHORIZED);
        }
        const token = authHeader.slice(7);
        try {
            const result = await this.authValidationService.validate(token);
            if (result.valid) {
                (req as Request & { user: RequestUser }).user = { userId: result.userId!, roles: result.roles! };
                next();
            } else {
                throw new HttpException('Invalid token', HttpStatus.UNAUTHORIZED);
            }
        } catch (error) {
            if (error instanceof HttpException) throw error;
            if (error instanceof ProviderError) {
                throw new HttpException('Authentication service unavailable', HttpStatus.SERVICE_UNAVAILABLE);
            }
            this.logger.error(`Auth middleware error: ${error}`);
            throw new HttpException('Authentication failed', HttpStatus.UNAUTHORIZED);
        }
    }
}
```

### Frontend: auth.api.ts

```typescript
const AUTH_API_URL = import.meta.env.VITE_AUTH_API_URL;

export async function loginUser(email: string, password: string): Promise<LoginResponse> {
    const res = await fetch(`${AUTH_API_URL}/auth/login`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password }),
    });
    if (!res.ok) throw new Error(await extractErrorMessage(res));
    return res.json();
}

export async function refreshAccessToken(refreshToken: string): Promise<LoginResponse> {
    const res = await fetch(`${AUTH_API_URL}/auth/refresh`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ refreshToken }),
    });
    if (!res.ok) throw new Error('Refresh failed');
    return res.json();
}
```

### Frontend: fetchWithAuth.ts

```typescript
const ACCESS_TOKEN_KEY = 'sftp_access_token';
const REFRESH_TOKEN_KEY = 'sftp_refresh_token';

export async function fetchWithAuth(input: RequestInfo, init?: RequestInit): Promise<Response> {
    // 1. Proactive refresh: decode exp from JWT, refresh if < 60s remaining
    await proactiveRefreshIfNeeded();

    // 2. Inject token
    const token = localStorage.getItem(ACCESS_TOKEN_KEY);
    const headers = new Headers(init?.headers);
    if (token) headers.set('Authorization', `Bearer ${token}`);

    // 3. Make request
    const response = await fetch(input, { ...init, headers });

    // 4. Handle 401: attempt refresh + retry once
    if (response.status === 401) {
        const refreshed = await tryRefresh();
        if (!refreshed) {
            clearTokens();
            window.location.href = '/login';
            return response;
        }
        const newToken = localStorage.getItem(ACCESS_TOKEN_KEY);
        headers.set('Authorization', `Bearer ${newToken}`);
        return fetch(input, { ...init, headers });
    }

    return response;
}
```

### Frontend: AuthContext.tsx

```typescript
const AuthContext = createContext<AuthState>(null!);

export function AuthProvider({ children }: { children: ReactNode }) {
    const [isAuthenticated, setIsAuthenticated] = useState(
        () => !!localStorage.getItem('sftp_access_token')
    );
    const [isLoading, setIsLoading] = useState(false);

    const login = async (email: string, password: string) => {
        setIsLoading(true);
        const data = await loginUser(email, password);
        localStorage.setItem('sftp_access_token', data.accessToken);
        localStorage.setItem('sftp_refresh_token', data.refreshToken);
        setIsAuthenticated(true);
        setIsLoading(false);
    };

    const logout = () => {
        localStorage.removeItem('sftp_access_token');
        localStorage.removeItem('sftp_refresh_token');
        setIsAuthenticated(false);
    };

    const getAccessToken = () => localStorage.getItem('sftp_access_token');

    return (
        <AuthContext.Provider value={{ isAuthenticated, isLoading, login, logout, getAccessToken }}>
            {children}
        </AuthContext.Provider>
    );
}

export const useAuth = () => useContext(AuthContext);
```

### Frontend: LoginPage.tsx (using react-hook-form)

```typescript
export function LoginPage() {
    const { login } = useAuth();
    const navigate = useNavigate();
    const { register, handleSubmit, formState: { errors, isSubmitting }, setError } = useForm<LoginForm>();

    const onSubmit = async ({ email, password }: LoginForm) => {
        try {
            await login(email, password);
            navigate('/');
        } catch {
            setError('root', { message: 'Email o contraseña inválidos' });
        }
    };

    return (
        <form onSubmit={handleSubmit(onSubmit)}>
            <input {...register('email', { required: true })} type="email" placeholder="Email" />
            <input {...register('password', { required: true })} type="password" placeholder="Contraseña" />
            {errors.root && <p>{errors.root.message}</p>}
            <button type="submit" disabled={isSubmitting}>Ingresar</button>
        </form>
    );
}
```
